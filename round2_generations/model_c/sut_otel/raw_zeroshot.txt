package com.example;

import java.net.http.HttpResponse;
import org.junit.jupiter.api.Test;

public class OTelShop_ModelC_ZeroShot_RobustnessTest extends OTelShopBaseTest {

  @Test
  void test_R1_GetProduct_InvalidId() throws Exception {
    HttpResponse<String> response = get("/api/products/invalid");
    assertNoServerError(response);
  }

  @Test
  void test_R1_GetCart_InvalidSessionId() throws Exception {
    HttpResponse<String> response = get("/api/cart?sessionId=invalid");
    assertNoServerError(response);
  }

  @Test
  void test_R1_PostCart_MissingUserId() throws Exception {
    String body = "{\"item\":{\"productId\":\"OLJCESPC7Z\",\"quantity\":1}}";
    HttpResponse<String> response = postJson("/api/cart", body);
    assertNoServerError(response);
  }

  @Test
  void test_R1_PostCart_InvalidProductId() throws Exception {
    String body = "{\"userId\":\"user123\",\"item\":{\"productId\":\"invalid\",\"quantity\":1}}";
    HttpResponse<String> response = postJson("/api/cart", body);
    assertNoServerError(response);
  }

  @Test
  void test_R1_DeleteCart_MissingSessionId() throws Exception {
    HttpResponse<String> response = delete("/api/cart");
    assertNoServerError(response);
  }

  @Test
