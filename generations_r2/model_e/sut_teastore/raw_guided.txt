package com.example;

import java.net.http.HttpResponse;
import org.junit.jupiter.api.Test;

public class TeaStore_ModelE_Guided_RobustnessTest extends TeaStoreBaseTest {

    @Test
    public void test_R1_EmptyCategory() throws Exception {
        HttpResponse<String> response = get("/tools.descartes.teastore.webui/category?id=");
        assertNoServerError(response);
    }

    @Test
    public void test_R2_NullCategory() throws Exception {
        HttpResponse<String> response = get("/tools.descartes.teastore.webui/category?id=null");
        assertNoServerError(response);
    }

    @Test
    public void test_R3_NegativeOneProduct() throws Exception {
        HttpResponse<String> response = get("/tools.descartes.teastore.webui/product?id=-1");
        assertNoServerError(response);
    }

    @Test
    public void test_R4_MinIntProductId() throws Exception {
        HttpResponse<String> response = get("/tools.descartes.teastore.webui/product?id=" + Integer.MIN_VALUE);
        assertNoServerError(response);
    }

    @Test
    public void test_R5_MaxIntPlusOneProduct() throws Exception {
        HttpResponse<String> response = get("/tools.descartes.teastore.webui/product?id=" + (Integer.MAX_VALUE + 1));
        assertNoServerError(response);
    }

    @Test
    public void test_R6_LoginWithNullPassword() throws Exception {
        HttpResponse<String> response = post("/tools.descartes.teastore.webui/loginAction", "username=user&password=null");
        assertNoServerError(response);
    }

    @Test
    public void test_R7_AlphanumericProductIdInCart() throws Exception {
        HttpResponse<String> response = post("/tools.descartes.teastore.webui/cartAction", "addToCart=&productid=abc123");
        assertNoServerError(response);
    }

    @Test
    public void test_R8_AddNonPrintableCharsToUsername() throws Exception {
        String nonPrintableChars = "\u0001\u0002\u0003";
        HttpResponse<String> response = post("/tools.descartes.teastore.webui/loginAction", "username=" + nonPrintableChars + "&password=pwd");
        assertNoServerError(response);
    }

    @Test
    public void test_R9_RemoveProductFromCartWithNullId() throws Exception {
        HttpResponse<String> response = post("/tools.descartes.teastore.webui/cartAction", "removeProduct=&productid=null");
        assertNoServerError(response);
    }
}
